# THE ONLY FILE IN THIS DIRECTORY THAT STILL INSTALLS TENSORFLOW.
#
# WHO: whoever converts a Keras (HDF5) model to ONNX with
#      scripts/convert_models_to_onnx.py, and docker/ci/Dockerfile's first
#      stage, which builds the ONNX test fixture that way.
# WHEN: dev-only, in a THROWAWAY environment. The analyzer does not need it and
#      never imports TensorFlow — it runs the .onnx through onnxruntime.
#
#     pip install -r requirements/convert.txt
#
# Keras 2 HDF5 needs the legacy-Keras shim under TF >= 2.16: export
# TF_USE_LEGACY_KERAS=1 before TensorFlow is imported (the standalone entry
# point sets this itself).
tensorflow==2.16.2
tf-keras==2.16.0
tf2onnx==1.17.0
# Pin onnx to a version compatible with the ml-dtypes~=0.3.1 that TF 2.16.2
# pins. Newer onnx (>=1.18) imports ml_dtypes.float4_e2m1fn (needs ml-dtypes
# >=0.5) and fails to import in a clean TF env.
onnx==1.16.2
onnxruntime==1.19.2

# APPLE SILICON: `tensorflow==2.16.2` above is CPU-only on macOS/arm64. To let
# TensorFlow use the GPU via Metal — needed by tests/test_standalone_pipeline.py,
# which CI --ignores because it imports TensorFlow at module scope — add the
# PluggableDevice as well:
#
#     pip install -r requirements/convert.txt tensorflow-metal==1.2.0
#
# 1.2.0 is the newest release and is hard-pinned to TF ~2.16, NOT 2.17+, which
# is why the TF pin above must not move on that machine.
